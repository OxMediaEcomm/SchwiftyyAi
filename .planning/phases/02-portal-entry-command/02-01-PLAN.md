```js
const RESET       = '\x1b[0m';
```

**sleep helper:**
```js
function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}
```

**printFrame function:**
```js
function printFrame(lines, frameIndex) {
  if (frameIndex > 0) {
    process.stdout.write(CURSOR_UP(lines.length));
  }
  for (const line of lines) {
    process.stdout.write(CLEAR_LINE + line + '\n');
  }
}
```

**animate async function:**

2. Print frameHeight blank lines to anchor the cursor position
